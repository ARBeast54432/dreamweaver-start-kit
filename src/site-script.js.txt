
const $ = (s) => document.querySelector(s);
const root = document.documentElement;

/* ---------- theme (remembers the choice) ---------- */
function setTheme(t) {
  root.dataset.theme = t;
  $('#mode').textContent = t === 'dark' ? 'DAY MODE' : 'NIGHT MODE';
  try { localStorage.setItem('theme', t); } catch (e) {}
}
let saved = null;
try { saved = localStorage.getItem('theme'); } catch (e) {}
setTheme(saved === 'dark' ? 'dark' : 'light');
$('#mode').onclick = () => setTheme(root.dataset.theme === 'dark' ? 'light' : 'dark');

/* ---------- date logic: birthday page + countdown ---------- */
const BIRTH_YEAR = 2011; // she turns (current year - 2011) on 30 Nov
const now = new Date(), year = now.getFullYear();
const today = new Date(year, now.getMonth(), now.getDate());
let next = new Date(year, 10, 30);
if (today > next) next = new Date(year + 1, 10, 30);
const days = Math.round((next - today) / 86400000);
const isBirthday = (now.getMonth() === 10 && now.getDate() === 30) || new URLSearchParams(location.search).has('birthday'); // add ?birthday to preview
$('#yr').textContent = year;
const tk = '// EESHIKA JAKKA // GOALKEEPER // SCIENCE STREAM // 12 COUNTRIES // ' + (days === 0 ? 'BIRTHDAY TODAY' : 'BIRTHDAY IN ' + days + ' DAYS') + ' ';
$('#tk').textContent = tk + tk;
const mq = 'GOALKEEPER /// SCIENCE /// ASIAN CUISINE /// GOLDEN RETRIEVERS /// 12 COUNTRIES /// ';
$('#mqi').textContent = mq + mq;
if (isBirthday) {
  const age = year - BIRTH_YEAR;
  $('#bd-age').textContent = age; $('#bd-v').textContent = age;
  $('#bd-date').textContent = '30.11.' + year + ' // BIRTHDAY MODE';
  $('#bday').hidden = false; $('#site').hidden = true;
  $('#enter').onclick = () => { $('#bday').hidden = true; $('#site').hidden = false; window.scrollTo(0, 0); };
}

  $('#cards').innerHTML = cd.map((c, i) => {

/* ---------- game: she always wins ---------- */
const P = [[17, 25], [50, 25], [83, 25], [17, 75], [50, 75], [83, 75]];
const L = ['TOP LEFT', 'TOP MID', 'TOP RIGHT', 'LOW LEFT', 'LOW MID', 'LOW RIGHT'];
const Q = ['Better luck next time.', 'Not today.', 'Nice try. Really.', 'Was that your best?', 'Too easy.', 'I saw that coming.', 'Cute.', 'Try the other corner. Same result.', 'Next.', 'Zero goals. Still zero.'];
$('#zones').innerHTML = L.map((l, i) => '<button class="zone" data-z="' + i + '">' + l + '</button>').join('');
const gk = $('#gk'), ball = $('#ball'), speech = $('#speech');
let busy = false, saves = 0, last = -1, timer;
function rest() {
  busy = false;
  gk.style.left = '50%'; gk.style.top = '62%'; gk.style.transform = 'translate(-50%,-50%)';
  ball.style.left = '50%'; ball.style.top = '112%';
  speech.style.opacity = 0; $('#status').textContent = 'GO AGAIN.';
  document.querySelectorAll('.zone.hit').forEach((z) => z.classList.remove('hit'));
}
$('#zones').onclick = (e) => {
  const b = e.target.closest('.zone');
  if (!b || busy) return;
  busy = true;
  const z = +b.dataset.z, col = z % 3;
  gk.style.left = P[z][0] + '%'; gk.style.top = P[z][1] + '%';
  gk.style.transform = 'translate(-50%,-50%) rotate(' + (col === 0 ? -60 : col === 2 ? 60 : 0) + 'deg)';
  ball.style.left = P[z][0] + '%'; ball.style.top = P[z][1] + '%';
  b.classList.add('hit');
  let i = Math.floor(Math.random() * Q.length);
  if (i === last) i = (i + 1) % Q.length;
  last = i;
  speech.textContent = Q[i]; speech.style.opacity = 1;
  saves++; $('#saves').textContent = saves; $('#status').textContent = 'SAVED.';
  timer = setTimeout(rest, 1400);
};
$('#reset').onclick = () => { clearTimeout(timer); saves = 0; $('#saves').textContent = 0; rest(); $('#status').textContent = 'PICK A SPOT.'; };
