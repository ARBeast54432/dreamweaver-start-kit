/* ---------- one shared rAF loop: every effect below hangs off this ---------- */
const sched = (() => {
  const tasks = new Set(); let raf = 0, last = 0;
  function loop(t) {
    raf = 0;
    if (document.hidden) { last = 0; return; }              // hidden tab: nothing runs
    const dt = Math.min(.05, ((t - (last || t)) / 1000) || .016); last = t;
    tasks.forEach((fn) => {
      let keep = true;
      try { keep = fn(t, dt) !== false; } catch (e) { console.error(e); keep = false; }
      if (!keep) tasks.delete(fn);
    });
    if (tasks.size) raf = requestAnimationFrame(loop); else last = 0;
  }
  const kick = () => { if (!raf && tasks.size && !document.hidden) { last = 0; raf = requestAnimationFrame(loop); } };
  document.addEventListener('visibilitychange', () => { root.classList.toggle('ej-paused', document.hidden); kick(); });
  // add(fn): fn(t, dt) runs every frame until it returns false. add() returns a function that removes it.
  return { add(fn) { tasks.add(fn); kick(); return () => tasks.delete(fn); } };
})();
const rmq = matchMedia('(prefers-reduced-motion: reduce)');

/* ---------- one small toast ---------- */
const toast = (() => {
  let el = null, tm = 0;
  function hide() { if (el) el.classList.remove('show', 'act'); }
  return (text, o = {}) => {
    if (!el) {
      el = document.createElement('div'); el.id = 'ej-toast';
      el.setAttribute('role', 'status'); el.setAttribute('aria-live', 'polite');
      document.body.appendChild(el);
    }
    clearTimeout(tm); el.textContent = '';
    const s = document.createElement('span'); s.textContent = text; el.appendChild(s);
    const bs = o.buttons || [];
    bs.forEach((b) => {
      const k = document.createElement('button'); k.type = 'button'; k.textContent = b.label;
      k.onclick = () => { hide(); if (b.onClick) b.onClick(); };
      el.appendChild(k);
    });
    el.classList.toggle('act', bs.length > 0);
    const hd = document.querySelector('header');
    el.style.top = (Math.max(0, hd ? hd.getBoundingClientRect().bottom : 0) + 12) + 'px';
    el.classList.add('show');
    tm = setTimeout(hide, o.ms || 2200);
  };
})();

/* ---------- theme ---------- */
const THEME_KEY = 'theme';
const savedTheme = () => { try { const v = localStorage.getItem(THEME_KEY); return v === 'dark' || v === 'light' ? v : null; } catch (e) { return null; } };
const autoTheme = () => { const h = new Date().getHours(); return h >= 19 || h < 6 ? 'dark' : 'light'; };   // first visit: dark from 19:00 to 06:00
function paintTheme(t) {
  root.dataset.theme = t;
  const m = $('#mode'); if (!m) return;
  m.setAttribute('aria-label', t === 'dark' ? 'Switch to day mode' : 'Switch to night mode');
  const mt = m.querySelector('.mt'); if (mt) mt.textContent = t === 'dark' ? 'DAY MODE' : 'NIGHT MODE';
}
// Only the visitor calls this (toggle button or the terminal's night/day), so this is the only place a choice is saved.
function setTheme(t) {
  if (t !== 'dark' && t !== 'light') return;
  try { localStorage.setItem(THEME_KEY, t); } catch (e) {}
  if (root.dataset.theme === t) return;
  if (rmq.matches) { paintTheme(t); return; }
  const m = $('#mode'), r = m ? m.getBoundingClientRect() : null;
  const x = r && r.width ? r.left + r.width / 2 : innerWidth - 40, y = r && r.height ? r.top + r.height / 2 : 40;
  if (document.startViewTransition) {                     // circular wipe growing out of the toggle button
    const vt = document.startViewTransition(() => paintTheme(t));
    vt.ready.then(() => {
      const rad = Math.hypot(Math.max(x, innerWidth - x), Math.max(y, innerHeight - y));
      root.animate({ clipPath: ['circle(0px at ' + x + 'px ' + y + 'px)', 'circle(' + rad + 'px at ' + x + 'px ' + y + 'px)'] },
        { duration: 650, easing: 'cubic-bezier(.4,0,.2,1)', pseudoElement: '::view-transition-new(root)' });
    }).catch(() => {});
    return;
  }
  // fallback: plain fade, out of the old background colour
  const veil = document.createElement('div');
  veil.style.cssText = 'position:fixed;inset:0;z-index:9998;pointer-events:none;opacity:1;transition:opacity .4s ease;background:' + getComputedStyle(document.body).backgroundColor;
  document.body.appendChild(veil);
  paintTheme(t);
  void veil.offsetWidth; veil.style.opacity = 0;
  setTimeout(() => veil.remove(), 520);
}
paintTheme(savedTheme() || autoTheme());                  // never saved here: no saved choice means "follow the clock"

// cheat on: the keeper dives the wrong way, to the far side of whatever you aimed at
function wrongWay(z) {
  const c = z % 3, row = Math.floor(Math.random() * 2);
  const col = c === 0 ? 2 : c === 2 ? 0 : (Math.random() < .5 ? 0 : 2);
  return row * 3 + col;
}

  gk.style.left = P[kz][0] + '%'; gk.style.top = P[kz][1] + '%';

  const layers=SPEC.map(s=>({k:s.k,big:s.r[1]>1.5,st:Array.from({length:s.n},()=>({x:rnd(),y:rnd(),r:s.r[0]+rnd()*(s.r[1]-s.r[0]),a:s.a[0]+rnd()*(s.a[1]-s.a[0]),ph:rnd()*6.283,sp:s.sp[0]+rnd()*(s.sp[1]-s.sp[0]),c:TINT[Math.floor(rnd()*TINT.length)]}))}));
  let W=0,H=0,dpr=1,sy=window.scrollY,siteH=0,stop=null,lastD=0,wait=12+Math.random()*13,fl=null,mp=-1;
  const dark=()=>root.dataset.theme==='dark';
  function size(){ dpr=Math.min(2,window.devicePixelRatio||1); W=window.innerWidth; H=window.innerHeight; cv.width=Math.round(W*dpr); cv.height=Math.round(H*dpr); }
  function draw(tm,still){
    cx.setTransform(dpr,0,0,dpr,0,0); cx.clearRect(0,0,W,H);
    const dens=Math.max(.5,Math.min(1,W*H/900000));             // a thinner sky on phones
    layers.forEach(L=>{
      const cut=Math.ceil(L.st.length*dens), shift=still?0:(sy*L.k)%H;
      for(let i=0;i<cut;i++){
        const s=L.st[i], x=s.x*W, y=(((s.y*H-shift)%H)+H)%H;    // wraps top to bottom, so it never runs out of sky
        cx.globalAlpha=s.a*(still?1:.68+.32*Math.sin(tm*s.sp+s.ph)); cx.fillStyle=s.c;
        cx.beginPath(); cx.arc(x,y,s.r,0,6.2832); cx.fill();
        if(L.big){ cx.globalAlpha*=.45; cx.fillRect(x-s.r*3.2,y-.4,s.r*6.4,.8); cx.fillRect(x-.4,y-s.r*3.2,.8,s.r*6.4); }
      }
    });
    cx.globalAlpha=1;
  }
  function placeMoon(p){ if(Math.abs(p-mp)<.0005) return; mp=p; moon.style.transform='translate3d('+(-p*14).toFixed(2)+'vw,'+(p*28).toFixed(2)+'vh,0)'; }
  function task(t,dt){
    placeMoon(Math.min(1,sy/Math.max(1,siteH-H)));              // the moon drifts slowly down and left as you scroll
    if(t-lastD>=33){ lastD=t; draw(t/1000,false); }
    wait-=dt;
    if(!fl&&wait<=0){                                           // a shooting star every 12 to 25 seconds
      const dir=Math.random()<.5?-1:1, th=(24+Math.random()*16)*Math.PI/180, dist=Math.min(W,620)*(.55+Math.random()*.35);
      fl={t:0,dur:.85+Math.random()*.3,hx:W*(.15+Math.random()*.7),hy:H*(.04+Math.random()*.38),dx:dir*Math.cos(th)*dist,dy:Math.sin(th)*dist,rot:(dir>0?th:Math.PI-th)*180/Math.PI};
    }
    if(fl){
      fl.t+=dt; const k=fl.t/fl.dur;
      if(k>=1){ shoot.style.opacity=0; fl=null; wait=12+Math.random()*13; }
      else{ const e=k*(2-k), o=k<.2?k/.2:1-(k-.2)/.8;
        shoot.style.transform='translate3d('+(fl.hx+fl.dx*e-140).toFixed(1)+'px,'+(fl.hy+fl.dy*e).toFixed(1)+'px,0) rotate('+fl.rot.toFixed(1)+'deg)';
        shoot.style.opacity=o.toFixed(2); }
    }
  }
  function sync(){
    const on=dark();
    if(on&&!rmq.matches){ size(); if(!stop) stop=sched.add(task); return; }
    if(stop){ stop(); stop=null; }
    fl=null; shoot.style.opacity=0;
    if(on){ size(); draw(0,true); placeMoon(0); }                // reduced motion: static stars, no drift, no shooting stars
  }
  window.addEventListener('scroll',()=>{ sy=window.scrollY; },{passive:true});
  window.addEventListener('resize',()=>{ if(!dark()) return; size(); if(!stop) draw(0,true); });
  if('ResizeObserver' in window&&site) new ResizeObserver((es)=>{ siteH=es[0].contentRect.height; }).observe(site);
  new MutationObserver(sync).observe(root,{attributes:true,attributeFilter:['data-theme']});
  if(rmq.addEventListener) rmq.addEventListener('change',sync);
  sync();
})();

/* ---------- header: hamburger menu at 700px and below ---------- */
(function(){
  const hdr=document.getElementById('hdr'), btn=document.getElementById('burger'), nav=document.getElementById('nav'); if(!hdr||!btn||!nav) return;
  const isOpen=()=>hdr.classList.contains('open');
  function set(open,focusFirst){
    hdr.classList.toggle('open',open); btn.setAttribute('aria-expanded',String(open)); btn.setAttribute('aria-label',open?'Close menu':'Open menu');
    if(open&&focusFirst){ const a=nav.querySelector('a'); if(a) a.focus(); }
  }
  btn.addEventListener('click',(e)=>set(!isOpen(),e.detail===0));          // e.detail === 0: opened from the keyboard, so move focus into the panel
  nav.addEventListener('click',(e)=>{ if(e.target.closest('a')) set(false); });   // closes after tapping a link
  document.addEventListener('keydown',(e)=>{ if(e.key==='Escape'&&isOpen()){ set(false); btn.focus(); } });
  document.addEventListener('pointerdown',(e)=>{ if(isOpen()&&!hdr.contains(e.target)) set(false); });
  matchMedia('(max-width:700px)').addEventListener('change',()=>set(false));

/* ---------- scroll reveal: slide up + fade, small stagger, once only ---------- */
(function(){
  if(!('IntersectionObserver' in window)||rmq.matches) return;
  const items=[...document.querySelectorAll('.sec > .lab, .sec > h2, .sec > p, .sec > .arena, .sec > .map, .sec > .term, #readout, #cards > .card, #stamps > .stamp, #faqs > .faq')];
  if(!items.length) return;
  const pending=new Set(items);
  function show(el,i){
    if(!pending.delete(el)) return;
    const d=Math.min(i*70,350);
    el.style.setProperty('--rd',d+'ms'); el.classList.add('in');
    setTimeout(()=>{ el.classList.remove('rv','in'); el.style.removeProperty('--rd'); },d+750);   // done: hand the element back to its normal hover styles
  }
  const io=new IntersectionObserver((es)=>{
    let i=0;
    es.forEach((e)=>{ if(e.isIntersecting||e.boundingClientRect.top<0){ io.unobserve(e.target); show(e.target,e.isIntersecting?i++:0); } });
  },{rootMargin:'0px 0px -6% 0px',threshold:.08});
  items.forEach((el)=>{ el.classList.add('rv'); io.observe(el); });
  // a jump (anchor link, restored scroll) can skip past things without ever intersecting: reveal whatever is now above the screen
  let st=0;
  window.addEventListener('scroll',()=>{ clearTimeout(st); st=setTimeout(()=>{ pending.forEach((el)=>{ if(el.getBoundingClientRect().bottom<0){ io.unobserve(el); show(el,0); } }); },160); },{passive:true});
})();

/* ---------- gentle card tilt (mouse only, max 4deg) ---------- */
(function(){
  if(!matchMedia('(hover:hover) and (pointer:fine)').matches||rmq.matches) return;
  const MAX=4; let el=null, px=0, py=0, queued=false;
  function reset(n){ if(!n) return; n.style.removeProperty('--rx'); n.style.removeProperty('--ry'); }
  function apply(){
    queued=false; if(!el) return false;
    const r=el.getBoundingClientRect(); if(!r.width||!r.height) return false;
    const fx=Math.max(0,Math.min(1,(px-r.left)/r.width)), fy=Math.max(0,Math.min(1,(py-r.top)/r.height));
    el.style.setProperty('--rx',((.5-fy)*2*MAX).toFixed(2)+'deg'); el.style.setProperty('--ry',((fx-.5)*2*MAX).toFixed(2)+'deg');
    return false;
  }
  document.addEventListener('pointermove',(e)=>{
    if(e.pointerType!=='mouse') return;
    const n=e.target.closest?e.target.closest('.card,.stamp'):null;
    if(n!==el){ reset(el); el=n; }
    if(!el) return;
    px=e.clientX; py=e.clientY;
    if(!queued){ queued=true; sched.add(apply); }       // one write per frame, on the shared loop
  },{passive:true});
  document.addEventListener('pointerleave',()=>{ reset(el); el=null; });
})();

/* ---------- little surprises: type "noodles", shake the phone ---------- */
(function(){
  const WORD='noodles'; let buf='', lastKey=0;
  const typing=(n)=>!!(n&&n.nodeType===1&&(n.isContentEditable||n.closest('input,textarea,select,[contenteditable=""],[contenteditable="true"]')));
  window.addEventListener('keydown',(e)=>{
    if(e.ctrlKey||e.metaKey||e.altKey||e.isComposing||e.repeat) return;
    if(typing(e.target)||typing(document.activeElement)){ buf=''; return; }     // never while an input is focused
    if(!e.key||e.key.length!==1) return;
    const now=Date.now(); if(now-lastKey>2000) buf=''; lastKey=now;
    buf=(buf+e.key.toLowerCase()).slice(-WORD.length);
    if(buf===WORD){ buf=''; const f=window.__ej&&window.__ej.noodles; if(f) f(); }
  });

  // shake => the dogs jump. Android and desktop just listen; iOS needs permission, asked politely on the first dog tap
  const DM=window.DeviceMotionEvent; if(!DM) return;
  let on=false, lx=null, ly=0, lz=0, hits=0, hitT=0, cool=0;
  function onMotion(e){
    const a=e.accelerationIncludingGravity; if(!a||a.x==null) return;
    const now=performance.now();
    if(lx!==null){
      const d=Math.abs(a.x-lx)+Math.abs(a.y-ly)+Math.abs(a.z-lz);
      if(d>26&&now>cool){
        if(now-hitT>700) hits=0;
        hits++; hitT=now;
        if(hits>=3){ hits=0; cool=now+2000; const f=window.__ej&&window.__ej.jumpAll; if(f) f(); }
      }
    }
    lx=a.x; ly=a.y; lz=a.z;
  }
  const attach=()=>{ if(on) return; on=true; window.addEventListener('devicemotion',onMotion,{passive:true}); };
  const needsAsk=typeof DM.requestPermission==='function';
  if(!needsAsk){ attach(); return; }
  const KEY='ejMotion'; let asked=false;
  const get=()=>{ try{return localStorage.getItem(KEY);}catch(e){return null;} };
  const put=(v)=>{ try{localStorage.setItem(KEY,v);}catch(e){} };
  function request(announce){
    try{
      DM.requestPermission().then((r)=>{
        if(r==='granted'){ put('yes'); attach(); if(announce) toast('Shake your phone and the dogs jump.',{ms:2600}); }
        else put('no');                                                       // denied: stay quiet
      }).catch(()=>{ put('no'); });
    }catch(e){}
  }
  window.__ej=Object.assign(window.__ej||{},{onDogTap(){
    if(on||asked) return; asked=true;
    const pref=get();
    if(pref==='yes'){ request(false); return; }                                    // said yes before: iOS just re-confirms
    if(pref) return;                                                          // asked or declined already: never nag
    put('asked');
    toast('Shake your phone to make the dogs jump? iOS will ask for motion access.',{ms:9000,buttons:[{label:'Sure',onClick:()=>request(true)},{label:'No thanks',onClick:()=>put('no')}]});
  }});
})();
